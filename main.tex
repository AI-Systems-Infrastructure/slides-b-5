\section{Introduction}

\begin{frame}{Recap of Module B.4: Cloud Deployment}{What is the cloud?}
  A computer or a specific computing service that we can access remotely over the internet.

  \begin{columns}[T]
  \begin{column}{0.58\textwidth}
    \begin{itemize}
      \item Usually, the hardware behind the cloud sits in \textbf{data centers}
      \item Technically, a cloud computer can be used for almost anything a personal computer can do
      \item The hardware and part of the software are owned and managed by someone else
    \end{itemize}
  \end{column}

  \begin{column}{0.4\textwidth}
    \begin{figure}[htpb]
        \begin{center}
            \includegraphics[width=\linewidth]{cern-data-centre}
            \caption*{Inside the CERN Data Centre, home to CERN's private cloud}
        \end{center}
    \end{figure}
  \end{column}
  \end{columns}

\end{frame}

\begin{frame}{Recap of Module B.4: Cloud Deployment}{Pros and cons of the cloud}
  \begin{block}{Benefits for us developers}
    \begin{itemize}
      \item A new computer can be set up and ready to use within minutes
      \item Pay for what we use, mitigate the need for purchasing hardware up front
      \item We do not need to worry about the management of the hardware, network infrastructure, electricity, etc.
    \end{itemize}
  \end{block}

  \begin{block}{Downsides for us developers}
    \begin{itemize}
      \item We never own the hardware and are subject to the cloud provider's pricing, terms, and conditions
      \item Higher possibility of data breach and privacy violation
      \item A system with a cloud component relies on a robust internet connection to be functional
    \end{itemize}
  \end{block}

\note{
}
\end{frame}


\begin{frame}{Module B.5: Edge Deployment}{Cases where we want to avoid cloud deployment}
  Scenarios where running our system on a computer we own and/or have physical access to fits better than on the cloud.

  \begin{itemize}
    \item A smart camera that has to react in milliseconds, where a cloud round trip is too slow
    \item An app processing medical scans we are not allowed to send to a third party
    \item A long-running service where the cumulative cloud bill has crossed the price of just buying a small server
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}{Module B.5: Edge Deployment}{Two related ideas}
    \begin{block}{Edge computing}
      Runs computation near where data is generated or where users interact with the system, instead of in a faraway data center.
      Motivated by \textbf{where} data and users are.
    \end{block}

    \begin{block}{Self-hosting}
        Runs services on hardware we own and control, instead of renting computers from a cloud provider.
        Motivated by \textbf{who} owns the hardware.
    \end{block}

    Not exactly the same, but they often overlap.

\note{
}
\end{frame}

    The edge is the boundary of the network, where data is collected or users interact with the system.

\begin{frame}{Edge computing}{Examples of edge computing}
  Data collection, user interaction, and compute are done on the same hardware.
  
  \begin{itemize}
    \item A surveillance camera detecting someone at the door
    \item A self-driving car deciding whether to brake
    \item A smartphone running on-device speech recognition
  \end{itemize}
\note{
}
\end{frame}


\begin{frame}{Edge computing}{Benefits of edge computing}
  \begin{itemize}
    \item \textbf{Latency:} the network round trip to a data center disappears
    \item \textbf{Bandwidth:} little to no data travels to the cloud
    \item \textbf{Privacy:} sensitive data never leaves the device
    \item \textbf{Robustness:} the system keeps working when the internet goes down
  \end{itemize}

\begin{frame}{Install operating system}

    \begin{itemize}

      \item \textbf{Edge \& Self-hosting:} We need to prepare the installation media and install the operating system ourselves

  \begin{block}{Common operating system picks}
    \begin{itemize}
      \item \textbf{Raspberry Pi:} \textit{Raspberry Pi Imager} writes a preconfigured SD card
      \item \textbf{Jetson:} \textit{JetPack SDK} ships an image with GPU drivers
      \item \textbf{Most x86 computers:} Linux installation media on a USB stick
    \end{itemize}
  \end{block}

\begin{frame}{CPU architecture}{A caveat to ``a container image works on any computer''}
  Within computers that adhere to the von Neumann architecture, the \textbf{CPU architecture} can still affect whether a container image is compatible with a specific computer.

  \begin{block}{Common CPU architectures}
    \begin{itemize}
      \item \textbf{x86-64} (\texttt{amd64}): almost every desktop, laptop, and most cloud computers
      \item \textbf{ARM64} (\texttt{aarch64}): smartphones, Raspberry Pi, Jetson, Apple Silicon Macs, and some cloud computers
      \item \textbf{RISC-V}: newer, open-source, mostly microcontrollers
    \end{itemize}
  \end{block}

  \vspace{1em}
  \live{Cloud computers with different CPU architectures on Hetzner.}
\note{
}
\end{frame}

\begin{frame}[fragile]{CPU architecture}{Building multi-architecture images}
    \begin{block}{Build on the target device}
      \textit{Directly building the image on the target edge device} guarantees the built image is compatible with the CPU architecture that edge device has.
    \end{block}

    \begin{block}{Build a multi-architecture image}
      Use \texttt{buildx} to build an image that is compatible with multiple CPU architectures.
        \begin{minted}{bash}
docker buildx build --platform linux/amd64,linux/arm64 \
    -t yourusername/my-ai-server:1.0 --push .
        \end{minted}
    \end{block}

\section{Remote Access}

\begin{frame}{Edge devices are usually not directly accessible over the internet}
  \begin{itemize}
    \item \textbf{Cloud computers:} usually come with a \textit{public IP address} that is directly accessible over the internet
    \item \textbf{Edge devices:} usually sit behind office or home local networks, and do not have a public IP address
  \end{itemize}

  \begin{figure}[htpb]
      \begin{center}
          \includegraphics[width=0.95\textwidth,height=0.55\textheight,keepaspectratio]{module-b5-nat-overview}
      \end{center}
  \end{figure}
\note{
}
\end{frame}


\begin{frame}{The prime suspect: NAT}
  \textbf{Network Address Translation (NAT)} lets every device on an office or home network share a single public IP.

  \begin{itemize}
    \item The router has a public IP from the ISP, shared by every device behind it
    \item The router gives each connected device a private IP like \texttt{192.168.1.42}
  \end{itemize}

  \begin{figure}[htpb]
      \begin{center}
          \includegraphics[width=0.95\textwidth,height=0.56\textheight,keepaspectratio]{module-b5-nat}
      \end{center}
  \end{figure}
\note{
}
\end{frame}


\begin{frame}{The prime suspect: NAT}
  \begin{itemize}
    \item \textbf{Outgoing requests work fine} since the router remembers the connection in a table
    \item \textbf{Incoming requests do not work} since the router has no idea where to send the request
  \end{itemize}

  \begin{figure}[htpb]
      \begin{center}
          \includegraphics[width=0.95\textwidth,height=0.6\textheight,keepaspectratio]{module-b5-nat-incoming}
      \end{center}
  \end{figure}
\note{
}
\end{frame}


\begin{frame}{The prime suspect: NAT}{Even worse: CGNAT}
  With \textbf{Carrier-grade NAT (CGNAT)}, the ISP itself does another layer of NAT in front of many customers. Even the router does not have a public IP address.

  \begin{figure}[htpb]
      \begin{center}
          \includegraphics[width=0.95\textwidth,height=0.66\textheight,keepaspectratio]{module-b5-cgnat}
      \end{center}
  \end{figure}
\note{
}
\end{frame}


\begin{frame}{Getting around NAT}
  Suppose your ISP does not enforce CGNAT and your home network has a public IP address:
  \begin{itemize}
    \item \textbf{Port forwarding} plus \textbf{dynamic DNS (DDNS)}
  \end{itemize}

  \vspace{1em}
  Works with or without CGNAT regardless:
  \begin{itemize}
    \item A cloud computer as a reverse proxy through a \textbf{VPN tunnel}
    \item Managed tunnel services
  \end{itemize}
\note{
}
\end{frame}


\begin{frame}{Getting around NAT}{Port forwarding}
  \begin{itemize}
    \item Your router has a public IP address. It just does not know where to forward the incoming requests.
    \item \textbf{Port forwarding} tells the router that incoming traffic on certain ports should go to a certain device.
  \end{itemize}

  \begin{figure}[htpb]
      \begin{center}
          \includegraphics[width=0.95\textwidth,height=0.55\textheight,keepaspectratio]{module-b5-port-forwarding}
      \end{center}
  \end{figure}
\note{
}
\end{frame}


\begin{frame}{Getting around NAT}{DDNS}
  \begin{itemize}
    \item Most public IP addresses the ISP gives us are dynamic and change every few weeks.
    \item \textbf{Dynamic DNS (DDNS)} is a service running on a computer that automatically updates the DNS record to reflect the latest public IP address of a home network.
  \end{itemize}

  \begin{figure}[htpb]
      \begin{center}
          \includegraphics[width=0.95\textwidth,height=0.6\textheight,keepaspectratio]{module-b5-ddns}
      \end{center}
  \end{figure}
\note{
}
\end{frame}


\begin{frame}{Getting around NAT}{Cloud computer as a reverse proxy}
  Incoming network traffic goes to a cloud computer first, and is then forwarded to the edge device through \textbf{VPN tunnels}.
  \begin{itemize}
    \item Compute is mostly done on the edge device, with the cloud computer only performing lightweight network traffic routing
    \item VPN tunnels themselves are fully encrypted, so any network traffic inside them is safe
  \end{itemize}

  \begin{figure}[htpb]

\end{frame}

\begin{frame}{Exercise: Self-host your AI server}
    Take the image from Module B.3. Host it on hardware you own. Try making it publicly accessible.

    \begin{block}{Outline}
        \begin{enumerate}
            \item Use your daily computer as the edge device, or use a ``real'' edge device if available
            \item Get it online, install Docker, run the container. Verify the server's reachability from another local machine
            \item Set up a VPN tunnel between the B.4 cloud computer and the device and point the Nginx proxy on the cloud VM to the edge device
